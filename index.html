1

一鍵傳送代碼綁定

2

子專案零負擔推播

自主查詢與解除

家長在 LINE 傳送「查詢」，中樞秒回已綁定學生狀態；傳送「解除綁定」即自主註銷，維護親師雙方權益。

💻 子專案快速接入代碼範本 (GAS / JS)

在子專案中複製此範例，即可立刻發送推播

// 1. 初始化中樞客戶端
var hub = new LineHubClient({
  hubUrl: "https://script.google.com/macros/s/您的中樞DeploymentId/exec",
  appId: "PROJECT_HW",
  appSecret: "sec_hw_demo_9823471"
});

// 2. 測試連線
var status = hub.testConnection();
Logger.log(status);

// 3. 發送推播通知給指定學生家長
var resp = hub.sendPushNotification({
  seatNo: "05",
  studentName: "王小明",
  title: "🚨 【今日作業缺交通知】",
  body: "國語生字語詞簿 第七課 尚未繳交，請今晚撥冗陪伴孩子補齊！",
  url: "https://example.com/portal?token=HW05"
});
Logger.log(resp);
